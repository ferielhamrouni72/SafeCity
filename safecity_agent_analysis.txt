"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import styles from "./agent.module.css";
import { AgentContext } from "./AgentContext";
import { useAgentSession } from "./useAgentSession";
import { useCompteurNouveaux } from "./useCompteurNouveaux";
import {
  IconAlert, IconBell, IconChart, IconGrid, IconLogout,
  IconMap, IconSearch, IconUser, Logo,
} from "./icons";

const NAV = [
  { href: "/agent/accueil", label: "Accueil", Icon: IconGrid, pret: true },
  { href: "/agent/incidents", label: "Incidents", Icon: IconAlert, pret: true },
  { href: "/agent/carte", label: "Carte", Icon: IconMap, pret: false },
  { href: "/agent/statistiques", label: "Statistiques", Icon: IconChart, pret: false },
  { href: "/agent/administration", label: "Administration", Icon: IconUser, pret: false },
];

const LIBELLES = { AGENT: "Agent superviseur", ADMINISTRATEUR: "Administrateur", CITOYEN: "Citoyen" };

export default function AgentShell({ children }: { children: React.ReactNode }) {
  const { session, deconnexion } = useAgentSession();
  const pathname = usePathname();
  const nouveaux = useCompteurNouveaux(session?.token ?? "");

  if (!session) return null;

  const { prenom, nom, role } = session.utilisateur;
  const initiales = `${prenom?.[0] ?? ""}${nom?.[0] ?? ""}`.toUpperCase();

  return (
    <div className={styles.shell}>
      <aside className={styles.sidebar}>
        <div className={styles.brand}>
          <Logo />
          <div>
            <p className={styles.brandName}>SafeCity</p>
            <p className={styles.brandSub}>SUPERVISION</p>
          </div>
        </div>

        <nav className={styles.nav} aria-label="Navigation principale">
          {NAV.map(({ href, label, Icon, pret }) => {
            const actif = pathname === href || pathname.startsWith(href + "/");
            if (!pret) {
              return (
                <span key={href} className={`${styles.navItem} ${styles.navOff}`} aria-disabled="true" title="Bientôt disponible">
                  <Icon />
                  {label}
                </span>
              );
            }
            return (
              <Link key={href} href={href} className={`${styles.navItem} ${actif ? styles.navActif : ""}`} aria-current={actif ? "page" : undefined}>
                <Icon />
                {label}
                {href === "/agent/incidents" && nouveaux > 0 && (
                  <span className={styles.badge} aria-label={`${nouveaux} nouveaux incidents`}>{nouveaux}</span>
                )}
              </Link>
            );
          })}
        </nav>

        <div className={styles.user}>
          <span className={styles.avatar}>{initiales}</span>
          <div className={styles.userInfo}>
            <p className={styles.userName}>{prenom} {nom}</p>
            <p className={styles.userRole}>{LIBELLES[role]}</p>
          </div>
          <button onClick={deconnexion} className={styles.logout} aria-label="Se déconnecter" title="Se déconnecter">
            <IconLogout />
          </button>
        </div>
      </aside>

      <div className={styles.contenu}>
        <header className={styles.topbar}>
          <div className={styles.search}>
            <IconSearch />
            <input type="search" placeholder="Rechercher…" aria-label="Rechercher" disabled />
          </div>
          <div className={styles.topRight}>
            <span className={styles.live}>
              <span className={styles.dot} />
              En direct
            </span>
            <button className={styles.bell} aria-label="Notifications"><IconBell /></button>
            <span className={styles.avatar}>{initiales}</span>
          </div>
        </header>

        <main className={styles.main}>
          <AgentContext.Provider value={session}>{children}</AgentContext.Provider>
        </main>
      </div>
    </div>
  );
}

export type Statut = "NOUVEAU" | "VALIDE" | "EN_COURS" | "RESOLU" | "REJETE" | "A_VERIFIER";
export type Gravite = "FAIBLE" | "MOYENNE" | "ELEVEE" | "CRITIQUE";

export const LIBELLE_STATUT: Record<Statut, string> = {
  NOUVEAU: "Nouveau",
  VALIDE: "Validé",
  EN_COURS: "En cours",
  RESOLU: "Résolu",
  REJETE: "Rejeté",
  A_VERIFIER: "À vérifier",
};

export const LIBELLE_GRAVITE: Record<Gravite, string> = {
  CRITIQUE: "Critique",
  ELEVEE: "Élevée",
  MOYENNE: "Moyenne",
  FAIBLE: "Faible",
};

export const COULEUR_GRAVITE: Record<Gravite, string> = {
  CRITIQUE: "#d64545",
  ELEVEE: "#e8833a",
  MOYENNE: "#e0a82e",
  FAIBLE: "#2f9e7a",
};

export const LIBELLE_TYPE: Record<string, string> = {
  ACCIDENT: "Accident",
  INCENDIE: "Incendie",
  INONDATION: "Inondation",
  FUITE_EAU: "Fuite d’eau",
  ARBRE_TOMBE: "Arbre tombé",
  NID_DE_POULE: "Nid-de-poule",
  INFRASTRUCTURE: "Infrastructure",
  AUTRE: "Autre",
};

export const COULEUR_NON_EVALUEE = "#c3cecf";

export function couleurGravite(g: Gravite | null): string {
  return g ? COULEUR_GRAVITE[g] : COULEUR_NON_EVALUEE;
}

// Passer à true quand la page correspondante existe : les liens deviennent cliquables
export const PAGES = { incidents: true, carte: false };

export function tempsRelatif(iso: string, maintenant: number): string {
  const t = new Date(iso);
  if (Number.isNaN(t.getTime())) return "";
  const minutes = Math.floor((maintenant - t.getTime()) / 60000);
  if (minutes < 1) return "À l’instant";
  if (minutes < 60) return `Il y a ${minutes} min`;
  const heure = t.toLocaleTimeString("fr-FR", { hour: "2-digit", minute: "2-digit" });
  const jour = (d: Date) => new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
  const ecart = Math.round((jour(new Date(maintenant)) - jour(t)) / 86400000);
  if (ecart <= 0) return `Aujourd’hui, ${heure}`;
  if (ecart === 1) return `Hier, ${heure}`;
  return `${t.toLocaleDateString("fr-FR", { day: "2-digit", month: "2-digit" })}, ${heure}`;
}

"use client";

import { useState, type CSSProperties } from "react";
import Link from "next/link";
import { useAgent } from "../AgentContext";
import { IconAlert, IconChevron, IconPin, IconSearch } from "../icons";
import StatutPastille from "../StatutPastille";
import {
  LIBELLE_GRAVITE, LIBELLE_STATUT, LIBELLE_TYPE,
  couleurGravite, tempsRelatif, type Gravite, type Statut,
} from "../incidentUi";
import styles from "./incidents.module.css";
import { useIncidents } from "./useIncidents";

const ONGLETS: { cle: Statut | ""; label: string }[] = [
  { cle: "", label: "Tous" },
  { cle: "NOUVEAU", label: "Nouveau" },
  { cle: "EN_COURS", label: "En cours" },
  { cle: "A_VERIFIER", label: "À vérifier" },
  { cle: "RESOLU", label: "Résolu" },
];

const STATUTS = Object.keys(LIBELLE_STATUT) as Statut[];
const GRAVITES = Object.keys(LIBELLE_GRAVITE) as Gravite[];

function IconFiltre() {
  return (
    <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
      <path d="M22 3H2l8 9.46V19l4 2v-8.54L22 3z" />
    </svg>
  );
}

export default function IncidentsPage() {
  const session = useAgent();
  const [date] = useState(() =>
    new Date().toLocaleDateString("fr-FR", { day: "numeric", month: "long" })
  );
  const [filtresOuverts, setFiltresOuverts] = useState(false);
  const { filtres, modifier, donnees, maintenant, erreur, chargement, totalPages } =
    useIncidents(session?.token ?? "");

  if (!session) return null;

  const incidents = donnees?.incidents ?? [];
  const filtresActifs = [filtres.gravite, filtres.type].filter(Boolean).length;

  return (
    <>
      <div className={styles.entete}>
        <div>
          <h1 className={styles.titre}>Incidents</h1>
          <p className={styles.sous}>Suivez et traitez les signalements.</p>
        </div>
        <p className={styles.date}>Aujourd’hui, {date}</p>
      </div>

      {erreur && <p role="alert" className={styles.erreur}>{erreur}</p>}

      <section className={styles.bloc}>
        <div className={styles.outils}>
          <label className={styles.recherche}>
            <IconSearch />
            <input
              type="search"
              value={filtres.q}
              onChange={(e) => modifier({ q: e.target.value })}
              placeholder="Rechercher un incident…"
              aria-label="Rechercher un incident"
            />
          </label>
          <button
            type="button"
            className={styles.boutonFiltres}
            onClick={() => setFiltresOuverts((v) => !v)}
            aria-expanded={filtresOuverts}
          >
            <IconFiltre />
            Filtres{filtresActifs > 0 ? ` (${filtresActifs})` : ""}
          </button>
        </div>

        {filtresOuverts && (
          <div className={styles.panneau}>
            <label className={styles.champ}>
              Statut
              <select value={filtres.statut} onChange={(e) => modifier({ statut: e.target.value as Statut | "" })}>
                <option value="">Tous</option>
                {STATUTS.map((s) => <option key={s} value={s}>{LIBELLE_STATUT[s]}</option>)}
              </select>
            </label>
            <label className={styles.champ}>
              Gravité
              <select value={filtres.gravite} onChange={(e) => modifier({ gravite: e.target.value as Gravite | "" })}>
                <option value="">Toutes</option>
                {GRAVITES.map((g) => <option key={g} value={g}>{LIBELLE_GRAVITE[g]}</option>)}
              </select>
            </label>
            <label className={styles.champ}>
              Type
              <select value={filtres.type} onChange={(e) => modifier({ type: e.target.value })}>
                <option value="">Tous</option>
                {Object.entries(LIBELLE_TYPE).map(([k, v]) => <option key={k} value={k}>{v}</option>)}
              </select>
            </label>
            <button
              type="button"
              className={styles.reinit}
              onClick={() => modifier({ statut: "", gravite: "", type: "", q: "" })}
            >
              Réinitialiser
            </button>
          </div>
        )}

        <div className={styles.onglets} role="tablist" aria-label="Filtrer par statut">
          {ONGLETS.map(({ cle, label }) => {
            const actif = filtres.statut === cle;
            const nombre = donnees?.compteurs[cle === "" ? "TOUS" : cle];
            return (
              <button
                key={label}
                type="button"
                role="tab"
                aria-selected={actif}
                className={`${styles.onglet} ${actif ? styles.ongletActif : ""}`}
                onClick={() => modifier({ statut: cle })}
              >
                {label}
                {nombre !== undefined && <span className={styles.compteur}>{nombre}</span>}
              </button>
            );
          })}
        </div>

        {chargement ? (
          <p className={styles.vide}>Chargement…</p>
        ) : incidents.length === 0 ? (
          <p className={styles.vide}>Aucun incident ne correspond à votre recherche.</p>
        ) : (
          <ul className={styles.liste}>
            {incidents.map((i) => (
              <li key={i.id}>
                <Link href={`/agent/incidents/${i.id}`} className={styles.ligne}>
                  <span className={styles.icone} style={{ "--c": couleurGravite(i.gravite) } as CSSProperties}>
                    <IconAlert />
                  </span>
                  <div className={styles.texte}>
                    <p className={styles.titreLigne}>{i.titre}</p>
                    <p className={styles.lieu}>
                      <IconPin />
                      {i.lieu ?? LIBELLE_TYPE[i.type] ?? "Lieu non précisé"} • #{i.id}
                    </p>
                  </div>
                  <div className={styles.droite}>
                    <span className={styles.temps}>{tempsRelatif(i.dateCreation, maintenant)}</span>
                    <StatutPastille statut={i.statut} />
                  </div>
                  <span className={styles.chevron}><IconChevron /></span>
                </Link>
              </li>
            ))}
          </ul>
        )}

        {totalPages > 1 && (
          <div className={styles.pagination}>
            <button type="button" disabled={filtres.page <= 1} onClick={() => modifier({ page: filtres.page - 1 })}>
              Précédent
            </button>
            <span>Page {filtres.page} sur {totalPages}</span>
            <button type="button" disabled={filtres.page >= totalPages} onClick={() => modifier({ page: filtres.page + 1 })}>
              Suivant
            </button>
          </div>
        )}
      </section>
    </>
  );
}

import { useCallback, useEffect, useRef, useState } from "react";
import { useRouter } from "next/navigation";
import { clearSession } from "@/lib/session";
import type { Gravite, Statut } from "../incidentUi";

export type LigneIncident = {
  id: number;
  titre: string;
  type: string;
  lieu: string | null;
  statut: Statut;
  gravite: Gravite | null;
  priorite: "NORMALE" | "URGENTE";
  dateCreation: string;
};

export type Compteurs = Record<"TOUS" | Statut, number>;

type Reponse = {
  incidents: LigneIncident[];
  total: number;
  page: number;
  limite: number;
  compteurs: Compteurs;
};

export type Filtres = {
  statut: Statut | "";
  gravite: Gravite | "";
  type: string;
  q: string;
  page: number;
};

const INITIAL: Filtres = { statut: "", gravite: "", type: "", q: "", page: 1 };
const LIMITE = 20;
const INTERVALLE_MS = 15000;

export function useIncidents(token: string) {
  const router = useRouter();
  const [filtres, setFiltres] = useState<Filtres>(INITIAL);
  const [qEffective, setQEffective] = useState("");
  const [donnees, setDonnees] = useState<Reponse | null>(null);
  const [maintenant, setMaintenant] = useState(0);
  const [erreur, setErreur] = useState("");
  const [chargement, setChargement] = useState(true);
  const sequence = useRef(0);

  // Toute modification de filtre ramène à la page 1 (sauf le changement de page lui-même)
  const modifier = useCallback((partiel: Partial<Filtres>) => {
    setFiltres((f) => ({ ...f, page: 1, ...partiel }));
  }, []);

  // Recherche : on attend la fin de la saisie avant d'interroger le serveur
  useEffect(() => {
    const t = setTimeout(() => setQEffective(filtres.q.trim()), 300);
    return () => clearTimeout(t);
  }, [filtres.q]);

  const { statut, gravite, type, page } = filtres;

  const charger = useCallback(async () => {
    if (!token) return;
    const numero = ++sequence.current; // ignore les réponses arrivées en retard
    const p = new URLSearchParams({ page: String(page), limite: String(LIMITE) });
    if (statut) p.set("statut", statut);
    if (gravite) p.set("gravite", gravite);
    if (type) p.set("type", type);
    if (qEffective) p.set("q", qEffective);

    try {
      const res = await fetch(`/api/agent/incidents?${p.toString()}`, {
        headers: { Authorization: `Bearer ${token}` },
        cache: "no-store",
      });
      if (numero !== sequence.current) return;
      if (res.status === 401) {
        clearSession();
        router.replace("/connexion");
        return;
      }
      if (res.status === 403) {
        setErreur("Accès réservé aux agents.");
        return;
      }
      if (!res.ok) {
        setErreur("Impossible de charger les incidents.");
        return;
      }
      setDonnees(await res.json());
      setMaintenant(Date.now());
      setErreur("");
    } catch {
      if (numero === sequence.current) setErreur("Serveur injoignable. Nouvelle tentative automatique.");
    } finally {
      if (numero === sequence.current) setChargement(false);
    }
  }, [token, statut, gravite, type, qEffective, page, router]);

  useEffect(() => {
    const premier = setTimeout(charger, 0);
    const timer = setInterval(() => {
      if (!document.hidden) charger();
    }, INTERVALLE_MS);
    return () => {
      clearTimeout(premier);
      clearInterval(timer);
    };
  }, [charger]);

  const totalPages = donnees ? Math.max(1, Math.ceil(donnees.total / donnees.limite)) : 1;

  return { filtres, modifier, donnees, maintenant, erreur, chargement, totalPages };
}

import { useCallback, useEffect, useRef, useState } from "react";
import { useRouter } from "next/navigation";
import { clearSession } from "@/lib/session";
import type { Gravite, Statut } from "../../incidentUi";

export type Commentaire = { id: number; texte: string; dateCreation: string; auteur: string };

export type IncidentDetail = {
  id: number;
  titre: string;
  type: string;
  description: string | null;
  lieu: string | null;
  latitude: number;
  longitude: number;
  statut: Statut;
  gravite: Gravite | null;
  priorite: "NORMALE" | "URGENTE";
  confianceIa: number | null;
  analyseIa: string | null;
  nombreTemoins: number;
  citoyenId: number;
  agent: string | null;
  dateCreation: string;
  datePriseEnCharge: string | null;
  dateResolution: string | null;
  photoIds: number[];
  commentaires: Commentaire[];
};

export type Action = "PRENDRE_EN_CHARGE" | "RESOUDRE" | "REJETER" | "ENREGISTRER";

const INTERVALLE_MS = 15000;

export function useIncident(token: string, id: number) {
  const router = useRouter();
  const [incident, setIncident] = useState<IncidentDetail | null>(null);
  const [maintenant, setMaintenant] = useState(0);
  const [introuvable, setIntrouvable] = useState(false);
  const [erreur, setErreur] = useState("");
  const [chargement, setChargement] = useState(true);
  const [envoi, setEnvoi] = useState(false);
  const occupe = useRef(false);

  const expirer = useCallback(() => {
    clearSession();
    router.replace("/connexion");
  }, [router]);

  const charger = useCallback(async () => {
    if (!token || !Number.isInteger(id) || id <= 0) {
      setIntrouvable(true);
      setChargement(false);
      return;
    }
    try {
      const res = await fetch(`/api/agent/incidents/${id}`, {
        headers: { Authorization: `Bearer ${token}` },
        cache: "no-store",
      });
      if (occupe.current) return; // une action est en cours : sa réponse fera foi
      if (res.status === 401) return expirer();
      if (res.status === 404) {
        setIntrouvable(true);
        return;
      }
      if (!res.ok) {
        setErreur("Impossible de charger l’incident.");
        return;
      }
      setIncident(await res.json());
      setMaintenant(Date.now());
      setErreur("");
    } catch {
      setErreur("Serveur injoignable. Nouvelle tentative automatique.");
    } finally {
      setChargement(false);
    }
  }, [token, id, expirer]);

  useEffect(() => {
    const premier = setTimeout(charger, 0);
    const timer = setInterval(() => {
      if (!document.hidden) charger();
    }, INTERVALLE_MS);
    return () => {
      clearTimeout(premier);
      clearInterval(timer);
    };
  }, [charger]);

  // Renvoie true si l'action a réussi
  const envoyer = useCallback(
    async (action: Action, extra: { gravite?: string; commentaire?: string }) => {
      occupe.current = true;
      setEnvoi(true);
      setErreur("");
      try {
        const res = await fetch(`/api/agent/incidents/${id}`, {
          method: "PATCH",
          headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
          body: JSON.stringify({ action, ...extra }),
        });
        if (res.status === 401) {
          expirer();
          return false;
        }
        if (res.status === 409) {
          occupe.current = false;
          await charger();
          setErreur("Cet incident vient d’être modifié par un autre agent. Les données ont été actualisées.");
          return false;
        }
        if (!res.ok) {
          const d = await res.json().catch(() => null);
          setErreur(d?.error?.message ?? "Action impossible.");
          return false;
        }
        setIncident(await res.json());
        setMaintenant(Date.now());
        return true;
      } catch {
        setErreur("Serveur injoignable.");
        return false;
      } finally {
        occupe.current = false;
        setEnvoi(false);
      }
    },
    [token, id, charger, expirer]
  );

  return { incident, maintenant, introuvable, erreur, chargement, envoi, envoyer };
}

// Liste des incidents (filtres, recherche, pagination) + compteurs par statut
export async function GET(req: Request) {
  const auth = await requireAuth(req, ["AGENT", "ADMINISTRATEUR"]);
  if ("response" in auth) return auth.response;

  const brut: Record<string, string> = {};
  for (const [k, v] of new URL(req.url).searchParams) if (v !== "") brut[k] = v;
  const analyse = Requete.safeParse(brut);
  if (!analyse.success) return erreur(400, "VALIDATION_ERROR", "Paramètres invalides");
  const { statut, gravite, type, q, page, limite } = analyse.data;

  const where: string[] = [];
  const params: (string | number)[] = [];
  if (statut) { where.push("statut = ?"); params.push(statut); }
  if (gravite) { where.push("gravite = ?"); params.push(gravite); }
  if (type) { where.push("type = ?"); params.push(type); }
  if (q) {
    const motif = `%${q.replace(/[\\%_]/g, "\\$&")}%`;
    const num = /^#?(\d{1,9})$/.exec(q);
    if (num) {
      where.push("(id = ? OR titre LIKE ? OR lieu LIKE ?)");
      params.push(Number(num[1]), motif, motif);
    } else {
      where.push("(titre LIKE ? OR lieu LIKE ?)");
      params.push(motif, motif);
    }
  }
  const clause = where.length ? `WHERE ${where.join(" AND ")}` : "";

  try {
    const [total, lignes, groupes] = await Promise.all([
      pool.query<RowDataPacket[]>(`SELECT COUNT(*) AS n FROM incident ${clause}`, params),
      pool.query<RowDataPacket[]>(
        `SELECT id, titre, type, lieu, statut, gravite, priorite,
                DATE_FORMAT(date_creation, '${ISO}') AS dateCreation
           FROM incident ${clause}
          ORDER BY date_creation DESC, id DESC
          LIMIT ? OFFSET ?`,
        [...params, limite, (page - 1) * limite]
      ),
      pool.query<RowDataPacket[]>("SELECT statut, COUNT(*) AS n FROM incident GROUP BY statut"),
    ]);

    const compteurs: Record<string, number> = { TOUS: 0 };
    for (const s of STATUTS) compteurs[s] = 0;
    for (const g of groupes[0]) {
      compteurs[String(g.statut)] = Number(g.n);
      compteurs.TOUS += Number(g.n);
    }

    return NextResponse.json({
      incidents: lignes[0],
      total: Number(total[0][0]?.n ?? 0),
      page,
      limite,
      compteurs,
    });
  } catch (e) {
    console.error("Échec du chargement des incidents :", e);
    return erreur(500, "SERVER_ERROR", "Erreur interne");
  }
}

function erreur(status: number, code: string, message: string) {

type Ctx = { params: Promise<{ id: string }> };

export async function GET(req: Request, ctx: Ctx) {

const Corps = z.object({
  action: z.enum(["PRENDRE_EN_CHARGE", "RESOUDRE", "REJETER", "ENREGISTRER"]),
  gravite: z.enum(GRAVITES).optional(),
  commentaire: z.string().trim().max(1000).optional(),
});

// Transitions autorisées : statut actuel -> nouveau statut
const TRANSITIONS = {
  PRENDRE_EN_CHARGE: { de: ["NOUVEAU", "VALIDE", "A_VERIFIER"], vers: "EN_COURS" },
  RESOUDRE: { de: ["EN_COURS"], vers: "RESOLU" },
  REJETER: { de: ["NOUVEAU", "VALIDE", "A_VERIFIER"], vers: "REJETE" },
} as const;

export async function PATCH(req: Request, ctx: Ctx) {
  const auth = await requireAuth(req, ["AGENT", "ADMINISTRATEUR"]);
  if ("response" in auth) return auth.response;

  const id = lireId((await ctx.params).id);
  if (!id) return erreur(404, "NOT_FOUND", "Incident introuvable");

  let json: unknown;
  try {
    json = await req.json();
  } catch {
    return erreur(400, "VALIDATION_ERROR", "Corps JSON invalide");
  }
  const analyse = Corps.safeParse(json);
  if (!analyse.success) return erreur(400, "VALIDATION_ERROR", "Données invalides");
  const corps = analyse.data;
  const transition = corps.action === "ENREGISTRER" ? null : TRANSITIONS[corps.action];

  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();

    // Verrou de ligne : si deux agents agissent en même temps, le second attend puis reçoit 409
    const [rows] = await conn.execute<RowDataPacket[]>(
      "SELECT statut, gravite FROM incident WHERE id = ? FOR UPDATE",
      [id]
    );
    const ancien = rows[0];
    if (!ancien) {
      await conn.rollback();
      return erreur(404, "NOT_FOUND", "Incident introuvable");
    }
    const statutActuel = String(ancien.statut);
    const termine = statutActuel === "RESOLU" || statutActuel === "REJETE";
    const autorise = transition
      ? (transition.de as readonly string[]).includes(statutActuel)
      : !termine;
    if (!autorise) {
      await conn.rollback();
      return erreur(409, "CONFLICT", "Cet incident a déjà été traité ou modifié");
    }

    const sets: string[] = [];
    const vals: (string | number)[] = [];
    if (transition) {
      sets.push("statut = ?");
      vals.push(transition.vers);
      if (corps.action === "PRENDRE_EN_CHARGE") {
        sets.push("agent_id = ?", "date_prise_en_charge = NOW()");
        vals.push(auth.user.id);
      } else if (corps.action === "RESOUDRE") {
        sets.push("agent_id = COALESCE(agent_id, ?)", "date_resolution = NOW()");
        vals.push(auth.user.id);
      }
    }
    const nouvelleGravite = corps.gravite && corps.gravite !== ancien.gravite ? corps.gravite : null;
    if (nouvelleGravite) {
      sets.push("gravite = ?");
      vals.push(nouvelleGravite);
    }
    const commentaire = corps.commentaire ?? "";

    if (sets.length === 0 && !commentaire) {
      await conn.rollback();
      return erreur(400, "VALIDATION_ERROR", "Rien à enregistrer");
    }

    if (sets.length > 0) {
      await conn.execute(`UPDATE incident SET ${sets.join(", ")} WHERE id = ?`, [...vals, id]);
    }
    if (transition) {
      await journaliser(
        {
          auteurId: auth.user.id,
          action: "CHANGEMENT_STATUT",
          typeEntite: "INCIDENT",
          idEntite: id,
          ancienneValeur: statutActuel,
          nouvelleValeur: transition.vers,
        },
        conn
      );
    }
    if (nouvelleGravite) {
      await journaliser(
        {
          auteurId: auth.user.id,
          action: "CHANGEMENT_GRAVITE",
          typeEntite: "INCIDENT",
          idEntite: id,
          ancienneValeur: (ancien.gravite as string | null) ?? null,
          nouvelleValeur: nouvelleGravite,
        },
        conn
      );
    }
    if (commentaire) {
      await conn.execute(
        "INSERT INTO incident_commentaire (incident_id, auteur_id, texte) VALUES (?, ?, ?)",
        [id, auth.user.id, commentaire]
      );
    }

    await conn.commit();
  } catch (e) {
    await conn.rollback().catch(() => {});
    console.error("Échec de la mise à jour de l'incident :", e);
    return erreur(500, "SERVER_ERROR", "Erreur interne");
  } finally {
    conn.release();
  }

  const incident = await chargerIncident(id);
  return NextResponse.json(incident);
}

export const UPLOAD_DIR = path.resolve(
  process.env.UPLOAD_DIR ?? path.join(process.cwd(), "uploads", "photos")

export type TypeImage = { mime: string; ext: string };

// On identifie le vrai format d'après les premiers octets du fichier,
// jamais d'après le nom ou le type envoyé par le client (falsifiable).
export function detecterTypeImage(b: Uint8Array): TypeImage | null {
  if (b.length >= 3 && b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff) {
    return { mime: "image/jpeg", ext: "jpg" };
  }
  const png = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a];
  if (b.length >= 8 && png.every((v, i) => b[i] === v)) {
    return { mime: "image/png", ext: "png" };
  }
  // WebP : "RIFF" .... "WEBP"
  if (
    b.length >= 12 &&
    b[0] === 0x52 && b[1] === 0x49 && b[2] === 0x46 && b[3] === 0x46 &&
    b[8] === 0x57 && b[9] === 0x45 && b[10] === 0x42 && b[11] === 0x50
  ) {
    return { mime: "image/webp", ext: "webp" };
  }
  return null;
}

import { NextResponse } from "next/server";
import type { RowDataPacket } from "mysql2";
import { pool } from "@/lib/db";
import { verifyToken, type Role } from "@/lib/auth";

export type AuthUser = { id: number; role: Role };

function erreur(status: number, code: string, message: string) {
  return NextResponse.json({ error: { code, message } }, { status });
}

export async function requireAuth(
  req: Request,
  roles?: Role[]
): Promise<{ user: AuthUser } | { response: NextResponse }> {
  const [scheme, token] = (req.headers.get("authorization") ?? "").split(" ");
  if (scheme !== "Bearer" || !token) {
    return { response: erreur(401, "UNAUTHENTICATED", "Authentification requise") };
  }

  let id: number;
  try {
    ({ id } = await verifyToken(token));
  } catch {
    return { response: erreur(401, "INVALID_TOKEN", "Token invalide ou expiré") };
  }

  const [rows] = await pool.execute<RowDataPacket[]>(
    "SELECT id, role, actif FROM utilisateur WHERE id = ?",
    [id]
  );
  const u = rows[0];
  if (!u) return { response: erreur(401, "INVALID_TOKEN", "Token invalide ou expiré") };
  if (!u.actif) return { response: erreur(403, "ACCOUNT_SUSPENDED", "Compte suspendu") };
  if (roles && !roles.includes(u.role as Role)) {
    return { response: erreur(403, "FORBIDDEN", "Accès refusé pour ce rôle") };
  }

  return { user: { id: u.id, role: u.role as Role } };
}

        if (e.target === e.currentTarget) onClose();
      }}
    >
      <div className={styles.fenetre} role="dialog" aria-modal="true" aria-label={`Photo de ${nom}`}>
        <div className={styles.entete}>
          <h3>Photo de {nom}</h3>
          <button ref={boutonFermer} onClick={onClose} className={styles.fermer}>
            Fermer
          </button>
        </div>
        <div className={styles.corps}>
          {chargement && <p className={styles.vide}>Chargement…</p>}
          {erreur && <p role="alert" className={styles.erreur}>{erreur}</p>}
          {url && <img src={url} alt={`Photo envoyée par ${nom}`} className={styles.image} />}
        </div>
      </div>
    </div>
  );
}

import { useCallback, useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { clearSession } from "@/lib/session";

export type PhotoRecue = {
  id: number;
  prenom: string;
  nom: string;
  dateReception: string;
};

const INTERVALLE_MS = 15000;

export function usePhotosRecues(token: string) {
  const router = useRouter();
  const [photos, setPhotos] = useState<PhotoRecue[]>([]);
  const [erreur, setErreur] = useState("");
  const [chargement, setChargement] = useState(true);

  const charger = useCallback(async () => {
    try {
      const res = await fetch("/api/agent/photos", {
        headers: { Authorization: `Bearer ${token}` },
        cache: "no-store",
      });
      if (res.status === 401) {
        clearSession();
        router.replace("/connexion");
        return;
      }
      if (res.status === 403) {
        setErreur("Accès réservé aux agents.");
        return;
      }
      if (!res.ok) {
        setErreur("Impossible de charger les photos.");
        return;
      }
      setPhotos(await res.json());
      setErreur("");
    } catch {
      setErreur("Serveur injoignable. Nouvelle tentative automatique.");
    } finally {
      setChargement(false);
    }
  }, [token, router]);

useEffect(() => {
  // Premier chargement via un minuteur : la règle ESLint interdit d'appeler
  // directement, dans l'effet, une fonction qui modifie un état
  const premier = setTimeout(charger, 0);
  const timer = setInterval(() => {
    if (!document.hidden) charger();
  }, INTERVALLE_MS);
  return () => {
    clearTimeout(premier);
    clearInterval(timer);
  };
}, [charger]);

  return { photos, erreur, chargement };
}
